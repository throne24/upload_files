2  DNS + DHCP
====================================================================

[整段] 2-1 写 /etc/named.conf
cat > /etc/named.conf <<'EOF'
options {
    listen-on port 53 { any; };
    listen-on-v6 port 53 { none; };
    directory "/var/named";
    allow-query { any; };
    recursion yes;
    dnssec-enable no;
    dnssec-validation no;
};

key "dhcpkey" {
    algorithm hmac-md5;
    secret "c2VjcmV0MTIzNDU2Nzg5MA==";
};

zone "wyulinuxos.com" IN {
    type master;
    file "wyulinuxos.com.zone";
    allow-update { key "dhcpkey"; };
};

zone "X.24.172.in-addr.arpa" IN {
    type master;
    file "X.24.172.rev";
    allow-update { key "dhcpkey"; };
};
EOF

[整段] 2-2 写正向区域
cat > /var/named/wyulinuxos.com.zone <<'EOF'
$TTL 86400
@   IN  SOA dns.wyulinuxos.com. root.wyulinuxos.com. (
        2025110401
        3600
        1800
        604800
        86400 )

    IN  NS  dns.wyulinuxos.com.
@   IN  A   172.24.X.1
dns IN  A   172.24.X.1

www   IN  A  172.24.X.1
samba IN  A  172.24.X.1
ftp   IN  A  172.24.X.1
mail  IN  A  172.24.X.1

course IN A  172.24.X.201

;3124001151 IN A 172.24.X.150
EOF

[整段] 2-3 写反向区域
cat > /var/named/X.24.172.rev <<'EOF'
$TTL 86400
@   IN  SOA dns.wyulinuxos.com. root.wyulinuxos.com. (
        2025110401
        3600
        1800
        604800
        86400 )

    IN  NS  dns.wyulinuxos.com.

1  IN  PTR dns.wyulinuxos.com.
1  IN  PTR www.wyulinuxos.com.
1  IN  PTR samba.wyulinuxos.com.
1  IN  PTR ftp.wyulinuxos.com.
1  IN  PTR mail.wyulinuxos.com.

201 IN  PTR course.wyulinuxos.com.

;150 IN PTR 3124001151.wyulinuxos.com.
EOF

[整段] 2-4 权限 + 检查 + 启动 named
chgrp named /var/named
chmod 775 /var/named
chown named:named /var/named/wyulinuxos.com.zone /var/named/X.24.172.rev
chmod 640 /var/named/wyulinuxos.com.zone /var/named/X.24.172.rev
restorecon -Rv /var/named
setsebool -P named_write_master_zones on

named-checkconf
named-checkzone wyulinuxos.com /var/named/wyulinuxos.com.zone
named-checkzone X.24.172.in-addr.arpa /var/named/X.24.172.rev

systemctl enable --now named
systemctl restart named

检查：三条检查命令都返回 OK；ss -lntup | grep :53 能看到 named。

[整段] 2-5 写 DHCP + DDNS
cat > /etc/dhcp/dhcpd.conf <<'EOF'
ddns-update-style interim;
ddns-updates on;
ddns-domainname "wyulinuxos.com.";
ddns-rev-domainname "in-addr.arpa.";
ignore client-updates;
authoritative;

option domain-name "wyulinuxos.com";
option domain-name-servers 172.24.X.1;
default-lease-time 600;
max-lease-time 7200;

key "dhcpkey" {
    algorithm hmac-md5;
    secret "c2VjcmV0MTIzNDU2Nzg5MA==";
};

zone wyulinuxos.com. {
    primary 172.24.X.1;
    key "dhcpkey";
}
zone X.24.172.in-addr.arpa. {
    primary 172.24.X.1;
    key "dhcpkey";
}

subnet 172.24.X.0 netmask 255.255.255.0 {
    range 172.24.X.150 172.24.X.200;
    option subnet-mask 255.255.255.0;
    option broadcast-address 172.24.X.255;
    option routers 172.24.X.1;
    option domain-name-servers 172.24.X.1;
}
EOF

[整段] 2-6 检查并启动 DHCP
dhcpd -t -cf /etc/dhcp/dhcpd.conf
systemctl enable --now dhcpd
systemctl restart dhcpd

检查：systemctl status dhcpd 是 active。

[整段] 2-7 客户端用学号拿地址
hostnamectl set-hostname 3124001151
cat > /etc/dhcp/dhclient.conf <<'EOF'
interface "ens33" {
    send host-name = gethostname();
    supersede domain-name "wyulinuxos.com";
    supersede domain-name-servers 172.24.X.1;
}
EOF

dhclient -r ens33
dhclient -v ens33

检查：ip a 拿到 172.24.X.150-200 的地址。

[整段] 2-8 nslookup 验收
nslookup www.wyulinuxos.com   172.24.X.1
nslookup samba.wyulinuxos.com 172.24.X.1
nslookup ftp.wyulinuxos.com   172.24.X.1
nslookup mail.wyulinuxos.com  172.24.X.1
nslookup 172.24.X.1 172.24.X.1
nslookup 3124001151.wyulinuxos.com 172.24.X.1

检查：四条 A 都是 172.24.X.1；学号返回 150-200 的地址。

    <Directory "/var/www/course">

    </Directory>
</VirtualHost>
EOF

5  Vsftpd + MySQL
====================================================================

[整段] 5-1 启动 MariaDB 并进入 mysql
systemctl enable --now mariadb
mysql -uroot

[SQL] 5-1 进入 mysql 后整段粘贴
CREATE DATABASE IF NOT EXISTS vsftpd DEFAULT CHARACTER SET utf8;
USE vsftpd;
DROP TABLE IF EXISTS users;
CREATE TABLE users (
    id       INT NOT NULL AUTO_INCREMENT,
    name     VARCHAR(32)  NOT NULL,
    passwd   VARCHAR(255) NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uk_name (name)
);
INSERT INTO users (name, passwd)
VALUES ('stone', PASSWORD('stone'));
SELECT name, passwd FROM users;

CREATE USER 'vsftpd'@'localhost' IDENTIFIED BY 'vftp123';
GRANT SELECT ON vsftpd.users TO 'vsftpd'@'localhost';
FLUSH PRIVILEGES;

检查：SELECT 出来 stone 的 passwd 以 * 开头。

[整段] 5-2 写 PAM
cat > /etc/pam.d/vsftpd.mysql <<'EOF'
auth    required pam_mysql.so user=vsftpd passwd=vftp123 host=localhost db=vsftpd table=users usercolumn=name passwdcolumn=passwd crypt=2
account required pam_mysql.so user=vsftpd passwd=vftp123 host=localhost db=vsftpd table=users usercolumn=name passwdcolumn=passwd crypt=2
EOF

[整段] 5-3 建映射账号 + stone 根目录
useradd -r -d /ftp -s /sbin/nologin virtual 2>/dev/null || true
mkdir -p /ftp/stone
chown -R virtual:virtual /ftp
chmod 755 /ftp /ftp/stone

[整段] 5-4 写 vsftpd.conf
cat > /etc/vsftpd/vsftpd.conf <<'EOF'
anonymous_enable=NO
local_enable=YES
write_enable=YES
local_umask=022
dirmessage_enable=YES

xferlog_enable=YES
xferlog_file=/var/log/vsftpd.log
xferlog_std_format=NO
log_ftp_protocol=YES

chroot_local_user=YES
allow_writeable_chroot=YES

guest_enable=YES
guest_username=virtual
virtual_use_local_privs=YES
pam_service_name=vsftpd.mysql
user_config_dir=/etc/vsftpd/user_conf

pasv_enable=YES
pasv_min_port=30000
pasv_max_port=30100

listen=YES
listen_ipv6=NO
secure_chroot_dir=/var/run/vsftpd/empty
userlist_enable=NO
tcp_wrappers=NO
EOF

[整段] 5-5 写 stone 独立配置
mkdir -p /etc/vsftpd/user_conf
cat > /etc/vsftpd/user_conf/stone <<'EOF'

[整段] 5-6 日志 + SELinux + 启动

mkdir -p /var/run/vsftpd/empty
