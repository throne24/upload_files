(1)开启服务
systemctl enable postfix
systemctl restart postfix

(2)修改配置文件 文件有就改 没有就加上
vi /etc/postfix/main.cf

mydestination = $mydomain, $myhostname 
mydomain = wyulinuxos.com 
myhostname = dns.wyulinuxos.com 
mynetworks = 172.24.90.0/24 
myorigin = $mydomain 
inet_interfaces = all	
relay_domains = wyulinuxos.com 
virtual_alias_domains = linux.com, wyulinuxos.com 
virtual_alias_maps = hash:/etc/postfix/virtual

#请确保 inet_interfaces = localhost 失效

(4)配置 cyrus-imapd 服务
vi /etc/postfix/main.cf

mailbox_transport = lmtp:unix:/var/lib/imap/socket/lmtp

退出；

systemctl start cyrus-imapd
systemctl enable cyrus-imapd
