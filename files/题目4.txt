(1)创建虚拟用户对应的系统用户
useradd -d /home/virtual -s /sbin/nologin	virtual
systemctl restart mysqld
mysql -u root -p
（无密码，直接回车进入）
create database ftpdb; 
use ftpdb;
create table	ftpusers(name char(20),password char(48));
insert into ftpusers(name,password) values ('stone',password('stone'));  
grant select on ftpdb.ftpusers to virtual@localhost identified by '123456';
select * from ftpusers;
quit
(显示stone的记录)

(2)修改vsftp主配置文件
vi	/etc/vsftpd/vsftpd.conf 
添加/修改：
pam_service_name=vsftpd.virtual 
guest_enable=YES
guest_username=virtual 
allow_writeable_chroot=YES
write_enable=YES
virtual_use_local_privs=YES
pasv_enable=YES
pasv_min_port=30000
pasv_max_port=31000
退出保存

(3)修改PAM认证文件
vi /etc/pam.d/vsftpd.virtual
auth required /usr/lib64/security/pam_mysql.so user=virtual passwd=123456 host=localhost db=ftpdb table=ftpusers usercolumn=name passwdcolumn=password crypt=2
account	required /usr/lib64/security/pam_mysql.so	user=virtual	passwd=123456 host=localhost db=ftpdb table=ftpusers	usercolumn=name	passwdcolumn=password crypt=2 
退出保存

(4)配置虚拟用户独立目录
vi /etc/vsftpd/vsftpd.conf
追加：
user_config_dir=/etc/vsftpd/vsftpd_user_conf
退出保存
mkdir -p	/etc/vsftpd/vsftpd_user_conf
mkdir /home/stone
chown -R virtual:virtual /home/stone/
setenforce 0
cd /etc/vsftpd/vsftpd_user_conf
vi stone
local_root=/home/stone
anon_upload_enable=YES	
write_enable=YES	
anon_mkdir_write_enable=YES	
anon_umask=022	
anon_world_readable_only=NO
退出保存
systemctl restart vsftpd

(5)配置日志记录
vi /etc/vsftpd/vsftpd.conf
修改参数：
xferlog_enable=YES
xferlog_std_format=NO
vsftpd_log_file=/var/log/vsftpd.log
退出保存
systemctl restart vsftpd
